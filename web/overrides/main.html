{# Theme override for ml-theses.org (mkdocs.yml: theme.custom_dir). Adds Open Graph and Twitter Card metadata so link previews (Slack, LinkedIn, WhatsApp, Mastodon, ...) show the current site name, description and card image instead of whatever the scraper guessed from the page body. Without these tags, preview services fall back to + <meta name="description"> and cache the result for a long time. #} {% extends "base.html" %} {% block extrahead %} {% set og_title = page.meta.title if page and page.meta and page.meta.title else (page.title if page and page.title and not page.is_homepage else config.site_name) %} {% if page and not page.is_homepage %} {% set og_title = og_title ~ " – " ~ config.site_name %} {% endif %} {% set og_description = page.meta.description if page and page.meta and page.meta.description else config.site_description %} {% set og_image = config.site_url ~ "assets/social-card.png" %} <meta property="og:type" content="website"> <meta property="og:site_name" content="{{ config.site_name }}"> <meta property="og:title" content="{{ og_title }}"> <meta property="og:description" content="{{ og_description }}"> <meta property="og:url" content="{{ config.site_url if page and page.is_homepage else page.canonical_url }}"> <meta property="og:image" content="{{ og_image }}"> <meta property="og:image:width" content="1200"> <meta property="og:image:height" content="630"> <meta property="og:image:alt" content="{{ config.site_name }}: a guide to doing a thesis in applied machine learning"> <meta name="twitter:card" content="summary_large_image"> <meta name="twitter:title" content="{{ og_title }}"> <meta name="twitter:description" content="{{ og_description }}"> <meta name="twitter:image" content="{{ og_image }}"> {% endblock %}
